% ─────────────────────────────────────────────────────────────────────────────
% preamble.tex  —  XeLaTeX, Times New Roman, IFAG A4 formatting
% ─────────────────────────────────────────────────────────────────────────────

% Fonts via fontspec (XeLaTeX)
\usepackage{fontspec}
\setmainfont{Times New Roman}
\setsansfont{Arial}
\setmonofont[Scale=0.88]{Courier New}

% Page: A4, IFAG margins — 2.5 cm all four sides (§1.2 fascicule)
\usepackage[a4paper,top=2.5cm,bottom=2.5cm,left=2.5cm,right=2.5cm]{geometry}

% Spacing: 1.5 lines
\usepackage{setspace}
\onehalfspacing

% Page numbers bottom-right, no header rule
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyfoot[R]{\thepage}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}
\fancypagestyle{plain}{\fancyhf{}\fancyfoot[R]{\thepage}\renewcommand{\headrulewidth}{0pt}}

% Chapter/section title sizes
\usepackage{titlesec}
\titleformat{\chapter}[block]
  {\centering\bfseries\fontsize{14}{17}\selectfont}
  {Chapter \thechapter~:~}{0pt}{}
\titlespacing*{\chapter}{0pt}{0pt}{12pt}
\titleformat{name=\chapter,numberless}[block]
  {\centering\bfseries\fontsize{14}{17}\selectfont}{}
  {0pt}{}
\titlespacing*{name=\chapter,numberless}{0pt}{0pt}{12pt}
\titleformat{\section}
  {\bfseries\fontsize{13}{15}\selectfont}{\thesection}{1em}{}
\titlespacing*{\section}{0pt}{12pt}{6pt}
\titleformat{\subsection}
  {\bfseries\itshape\fontsize{12}{14}\selectfont}{\thesubsection}{1em}{}
\titlespacing*{\subsection}{0pt}{12pt}{6pt}

% Paragraphs
\setlength{\parindent}{1cm}
\setlength{\parskip}{6pt}

% Line discipline (supervisor notes): keep a paragraph's first/last line with the
% rest rather than stranding one line on another page, do not stretch vertical
% space to fill a page (which would open gaps between points), and let microtype
% pull characters into the margin so fewer lines stick out past the right edge.
\usepackage{microtype}
\widowpenalty=10000
\clubpenalty=10000
\raggedbottom
% Let LaTeX stretch a line slightly rather than push a word past the right margin.
\setlength{\emergencystretch}{3em}

% TOC depth — IFAG max 3 levels (1.1.1), so depth = 2
\setcounter{tocdepth}{2}
\setcounter{secnumdepth}{2}

% TOC chapter entry format — "Chapter N : Title" inline bold (matches IFAG example)
\usepackage{titletoc}
\usepackage{ifthen}
\titlecontents{chapter}
  [0em]
  {\addvspace{1.0em}\bfseries}
  {\ifthenelse{\equal{\thecontentslabel}{A}\OR\equal{\thecontentslabel}{B}}{%
    Annexe~\thecontentslabel~: }{Chapter~\thecontentslabel~: }}
  {}
  {\titlerule*[0.5pc]{.}\contentspage}

% Figures and tables
\usepackage{graphicx}
\usepackage{booktabs}
\usepackage{longtable}
\usepackage{caption}
% IFAG format: "Figure X.Y : Description" / "Table X.Y : Description" (colon, matches TOC)
\DeclareCaptionLabelSeparator{ifagcolon}{ : }
\captionsetup[figure]{position=below,font={small,it},labelfont=bf,justification=centering,labelsep=ifagcolon}
\captionsetup[table]{position=above,font={small,it},labelfont=bf,justification=centering,labelsep=ifagcolon}
\usepackage{chngcntr}
\counterwithin{figure}{chapter}
\counterwithin{table}{chapter}

% Math symbols (for \square checkboxes in questionnaire) and math (\tfrac, etc.)
\usepackage{amsmath}
\usepackage{amssymb}

% TikZ — for inline diagrams (architecture, ERD, flowcharts, sequence diagrams)
\usepackage{tikz}
\usetikzlibrary{arrows.meta, shapes.geometric, shapes.misc, positioning,
                fit, backgrounds, calc, decorations.pathreplacing}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

% Code listings
\usepackage{listings}
\usepackage{xcolor}
\definecolor{codebg}{RGB}{248,248,248}
\definecolor{codeframe}{RGB}{200,200,200}
\definecolor{codegreen}{RGB}{0,128,0}
\definecolor{codepurple}{RGB}{128,0,128}
\definecolor{codeblue}{RGB}{0,0,200}
\lstdefinestyle{mymemoire}{%
  backgroundcolor=\color{codebg},frame=single,framerule=0.4pt,rulecolor=\color{codeframe},
  basicstyle=\ttfamily\small\singlespacing,
  keywordstyle=\color{codeblue}\bfseries,
  stringstyle=\color{codegreen},
  commentstyle=\color{codepurple}\itshape,
  numbers=left,numberstyle=\tiny\color{gray},numbersep=6pt,
  breaklines=true,breakatwhitespace=true,showstringspaces=false,
  tabsize=2,captionpos=b}
\lstset{style=mymemoire}
\lstdefinelanguage{TypeScript}[]{Java}{
  morekeywords={const,let,var,async,await,interface,type,export,import,from,of,
                string,number,boolean,void,any,null,undefined,readonly,Promise}}

% Bibliography (numeric clickable style — IFAG §2.13 allows [1] or (Author, year); numeric is the
% conventional choice for IT theses. Combined with hyperref, every \parencite{...} renders as a
% clickable [N] that links to the entry below.)
\usepackage[backend=biber,style=numeric-comp,sortcites=true,sorting=none]{biblatex}
\addbibresource{back/bibliography.bib}

% Extra table and list packages
\usepackage{float}
\usepackage{tabularx}
\usepackage{multirow}
\usepackage{array}
\usepackage{enumitem}
\setlist[itemize]{itemsep=2pt,topsep=4pt}
\setlist[enumerate]{itemsep=2pt,topsep=4pt}

% Hyperlinks
\usepackage[colorlinks=true,linkcolor=black,citecolor=blue!60!black,urlcolor=blue,
  pdftitle={Mobile Match Algeria - Memoire de Fin d'Etudes},
  pdfauthor={AMRATE Riad}]{hyperref}

% Allow long URLs / identifiers (e.g. arXiv ids) to break across lines so they
% never run past the right margin; \urlstyle{same} keeps them in the body font.
\usepackage{xurl}
\urlstyle{same}

% Bidi — right-to-left support for Arabic abstract (must load after hyperref)
\usepackage{bidi}
\newfontfamily\arabicfont[Script=Arabic, Scale=1.05]{Arial}
